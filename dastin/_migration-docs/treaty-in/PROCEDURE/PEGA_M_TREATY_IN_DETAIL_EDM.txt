CREATE OR REPLACE PROCEDURE POOLDATA.PEGA_M_TREATY_IN_DETAIL_EDM(
DataPega IN CLOB, 
IDPega IN VARCHAR2, 
P_TREATYID  IN VARCHAR2,
P_TREATYYEAR    IN VARCHAR2,
P_SOBID IN VARCHAR2,
P_SOB   IN VARCHAR2,
P_CEDINGID  IN VARCHAR2,
P_CEDING    IN VARCHAR2,
P_TREATYCONTRACTNAME    IN VARCHAR2,
P_PROPORTIONTYPE    IN VARCHAR2,
P_TREATYTYPE    IN VARCHAR2,
P_TREATYGROUPID IN VARCHAR2,
P_TREATYGROUP   IN VARCHAR2,
P_CLASSOFBUSINESSID IN VARCHAR2,
P_CLASSOFBUSINESS   IN VARCHAR2,
P_LIMITCURRENCY IN VARCHAR2,
P_LIMITVALUE    IN NUMBER,
P_RETENTIONCURRENCY IN VARCHAR2,
P_RETENTIONVALUE    IN NUMBER,
P_EPICURRENCY   IN VARCHAR2,
P_EPIVALUE  IN NUMBER,
P_MDPCURRENCY   IN VARCHAR2,
P_MDPVALUE  IN NUMBER,
P_NETPREMICURRENCY  IN VARCHAR2,
P_NETPREMIVALUE IN NUMBER,
P_DEDUCTION1    IN NUMBER,
P_DEDUCTION2    IN NUMBER,
P_LAYERTYPE IN VARCHAR2,
P_LAYER IN VARCHAR2,
P_LAYERPARTTYPE IN VARCHAR2,
P_LAYERPART IN VARCHAR2,
P_INSTALLMENTNO IN VARCHAR2,
P_SHARECURRENCY IN VARCHAR2,
P_SHAREVALUE    IN NUMBER,
P_RIOGR IN NUMBER,
P_RIONR IN NUMBER,
P_RNM_SHARE IN NUMBER,
P_SPREADINGTYPEID IN VARCHAR2,
P_SPREADINGTYPE IN VARCHAR2,
P_QS_OR IN VARCHAR2,
P_QS_RI IN VARCHAR2,
P_BROKERAGE IN VARCHAR2,
P_CESSIONCURRENCY IN VARCHAR2,
P_CESSIONVALUE IN VARCHAR2,
P_CURRENCY_RSMD IN VARCHAR2,
P_CURRENCY_EQ IN VARCHAR2,
P_CURRENCY_FLOOD_JABO IN VARCHAR2,
P_CURRENCY_FLOOD_NATION IN VARCHAR2,
P_LIMIT_RSMD IN VARCHAR2,
P_LIMIT_EQ IN VARCHAR2,
P_LIMIT_FLOOD_JABO IN VARCHAR2,
P_LIMIT_FLOOD_NATION IN VARCHAR2,
P_SPL_LINE IN VARCHAR2,
P_QS_PCT   IN VARCHAR2,
ErrMsg OUT VARCHAR2, 
IDPegaOut OUT VARCHAR2, 
StsSave OUT VARCHAR2
)
AS
id_TreatyIn_ins M_TREATY_IN_DETAIL_EDM.ID%TYPE;
id_TreatyIn_out M_TREATY_IN_DETAIL_EDM.ID%TYPE;
id_site M_SITE_DATABASE.ID%TYPE;

-- Tambahan variabel untuk commencement dan termination
        v_COMMENCEMENT DATE;
        v_TERMINATION  DATE;

BEGIN

    IF IDPega = 'UnknownId' THEN

       BEGIN
          SELECT ID INTO id_site FROM POOLDATA.M_SITE_DATABASE WHERE CURRENT_SITE='1';
       EXCEPTION WHEN OTHERS THEN
          ROLLBACK;
          RETURN;
       END;
       
       -- Ambil COMMENCEMENT dan TERMINATION dari TREATY_IN
           BEGIN
              SELECT to_date(COMMENCEMENT,'YYYYMMDD'), to_date(TERMINATION,'YYYYMMDD')
              INTO v_COMMENCEMENT, v_TERMINATION
              FROM POOLDATA.TREATY_IN_EDM
              WHERE ID = P_TREATYID;
           EXCEPTION 
              WHEN NO_DATA_FOUND THEN
                 v_COMMENCEMENT := NULL;
                 v_TERMINATION := NULL;
              WHEN OTHERS THEN
                 v_COMMENCEMENT := NULL;
                 v_TERMINATION := NULL;
           END;

       BEGIN
          id_TreatyIn_ins := id_site || lpad(to_Char(POOLDATA.M_TREATY_IN_DETAIL_SEQ.nextval),6,'0');
       EXCEPTION WHEN OTHERS THEN
          ROLLBACK;
          RETURN;
       END;

       BEGIN
          INSERT INTO POOLDATA.M_TREATY_IN_DETAIL_EDM (ID, JSONDATA)
          VALUES (id_TreatyIn_ins, replace(DataPega,'UnknownId',id_TreatyIn_ins));

          INSERT INTO POOLDATA.TREATYINDETAILEDM(
              ID, TREATYID, TREATYYEAR, SOBID, SOB, CEDINGID, CEDING,
              TREATYCONTRACTNAME, PROPORTIONTYPE, TREATYTYPE, TREATYGROUPID,
              TREATYGROUP, CLASSOFBUSINESSID, CLASSOFBUSINESS, LIMITCURRENCY,
              LIMITVALUE, RETENTIONCURRENCY, RETENTIONVALUE, EPICURRENCY, EPIVALUE,
              MDPCURRENCY, MDPVALUE, NETPREMICURRENCY, NETPREMIVALUE, DEDUCTION1,
              DEDUCTION2, LAYERTYPE, LAYER, LAYERPARTTYPE, LAYERPART, INSTALLMENTNO,
              SHARECURRENCY, SHAREVALUE, RIOGR, RIONR, RNM_SHARE, SPREADINGTYPEID,
              SPREADINGTYPE, QS_OR, QS_RI, BROKERAGE, CESSIONCURRENCY, CESSIONVALUE,
              CURRENCY_RSMD, CURRENCY_EQ, CURRENCY_FLOOD_JABO, CURRENCY_FLOOD_NATION,
              LIMIT_RSMD, LIMIT_EQ, LIMIT_FLOOD_JABO, LIMIT_FLOOD_NATION,SPL_LINE,QS_PCT,
              COMMENCEMENT, TERMINATION
          )
          VALUES (
              id_TreatyIn_ins, P_TREATYID, P_TREATYYEAR, P_SOBID, P_SOB, P_CEDINGID, P_CEDING,
              P_TREATYCONTRACTNAME, P_PROPORTIONTYPE, P_TREATYTYPE, P_TREATYGROUPID,
              P_TREATYGROUP, P_CLASSOFBUSINESSID, P_CLASSOFBUSINESS, P_LIMITCURRENCY,
              P_LIMITVALUE, P_RETENTIONCURRENCY, P_RETENTIONVALUE, P_EPICURRENCY, P_EPIVALUE,
              P_MDPCURRENCY, P_MDPVALUE, P_NETPREMICURRENCY, P_NETPREMIVALUE, P_DEDUCTION1,
              P_DEDUCTION2, P_LAYERTYPE, P_LAYER, P_LAYERPARTTYPE, P_LAYERPART, P_INSTALLMENTNO,
              P_SHARECURRENCY, P_SHAREVALUE, P_RIOGR, P_RIONR, P_RNM_SHARE, P_SPREADINGTYPEID,
              P_SPREADINGTYPE, P_QS_OR, P_QS_RI, P_BROKERAGE, P_CESSIONCURRENCY, P_CESSIONVALUE,
              P_CURRENCY_RSMD, P_CURRENCY_EQ, P_CURRENCY_FLOOD_JABO, P_CURRENCY_FLOOD_NATION,
              P_LIMIT_RSMD, P_LIMIT_EQ, P_LIMIT_FLOOD_JABO, P_LIMIT_FLOOD_NATION,P_SPL_LINE,P_QS_PCT,
               v_COMMENCEMENT, v_TERMINATION
          );

          COMMIT;
          id_TreatyIn_out := id_TreatyIn_ins;
       EXCEPTION WHEN OTHERS THEN
          StsSave := 0;
          ROLLBACK;
          RETURN;
       END;

    END IF;

    StsSave := 1;
    IDPegaOut := id_TreatyIn_out;

EXCEPTION
    WHEN OTHERS THEN
        StsSave := 0;
        ROLLBACK;
        RETURN;
END;
/
